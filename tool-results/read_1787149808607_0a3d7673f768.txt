   112→function FeatureBarChart({ features }: { features: FeatureBar[] }) {
   113→  return (
   114→    <div className="space-y-2">
   115→      {features.map((f) => {
   116→        const pct = Math.min(100, Math.max(0, ((f.value + 2) / 4) * 100));
   117→        const isBull = f.value > 0.1;
   118→        const isBear = f.value < -0.1;
   119→        return (
   120→          <div key={f.label} className="flex items-center gap-3">
   121→            <span className="w-12 text-xs font-mono text-muted-foreground text-right">{f.label}</span>
   122→            <div className="flex-1 relative h-6 bg-muted rounded-sm overflow-hidden">
   123→              {/* Center line */}
   124→              <div className="absolute left-1/2 top-0 bottom-0 w-px bg-border z-10" />
   125→              {/* Fill from center */}
   126→              <div
   127→                className={`absolute top-0 h-full transition-all duration-300 ${
   128→                  isBull ? 'bg-emerald-500/70 left-1/2' : isBear ? 'bg-red-500/70 right-1/2' : 'bg-muted-foreground/20 left-[45%] w-[10%]'
   129→                }`}
   130→                style={{ width: `${Math.abs(pct - 50)}%` }}
   131→              />
   132→            </div>
   133→            <span className={`w-14 text-xs font-mono text-right ${!f.valid ? 'text-muted-foreground/50' : isBull ? 'text-emerald-400' : isBear ? 'text-red-400' : 'text-muted-foreground'}`}>
   134→              {f.valid ? f.value.toFixed(2) : '---'}
   135→            </span>
   136→          </div>
   137→        );
   138→      })}
   139→    </div>
   140→  );
   141→}

   143→function ScoreGauge({ score }: { score: number }) {
   144→  const angle = score * 90; // -90 to +90
   145→  const color = score > 0.2 ? 'text-emerald-400' : score < -0.2 ? 'text-red-400' : 'text-yellow-400';
   146→  return (
   147→    <div className="flex flex-col items-center gap-1">
   148→      <div className="relative w-28 h-28">
   149→        <svg viewBox="0 0 100 60" className="w-full h-full">
   150→          {/* Background arc */}
   151→          <path d="M 10 55 A 40 40 0 0 1 90 55" fill="none" stroke="currentColor" strokeWidth="6" className="text-muted/30" strokeLinecap="round" />
   152→          {/* Score arc */}
   153→          <path
   154→            d="M 10 55 A 40 40 0 0 1 90 55"
   155→            fill="none"
   156→            stroke="currentColor"
   157→            strokeWidth="6"
   158→            strokeLinecap="round"
   159→            className={color}
   160→            strokeDasharray={`${Math.abs(angle) * 1.4} 200`}
   161→            strokeDashoffset={angle > 0 ? 0 : 0}
   162→            transform={angle >= 0 ? '' : `scale(-1,1) translate(-100,0)`}
   163→            style={{ opacity: Math.abs(score) * 2 }}
   164→          />
   165→          {/* Needle */}
   166→          <line
   167→            x1="50" y1="55"
   168→            x2="50" y2="20"
   169→            stroke="currentColor"
   170→            strokeWidth="2"
   171→            className={color}
   172→            transform={`rotate(${angle}, 50, 55)`}
   173→          />
   174→          <circle cx="50" cy="55" r="3" fill="currentColor" className={color} />
   175→        </svg>
   176→      </div>
   177→      <span className={`text-2xl font-bold font-mono ${color}`}>{score.toFixed(3)}</span>
   178→      <span className="text-xs text-muted-foreground">Composite Score</span>
   179→    </div>
   180→  );
   181→}

   183→function SignalLed({ side, confidence, ts }: { side: 'BUY' | 'SELL' | null; confidence: number; ts: number }) {
   184→  if (!side) return <div className="flex items-center gap-2 text-muted-foreground"><Radio className="w-4 h-4" /> <span className="text-sm">No signal</span></div>;
   185→  const isBuy = side === 'BUY';
   186→  return (
   187→    <div className={`flex items-center gap-3 p-3 rounded-lg ${isBuy ? 'bg-emerald-500/10 border border-emerald-500/30' : 'bg-red-500/10 border border-red-500/30'}`}>
   188→      <div className={`w-3 h-3 rounded-full animate-pulse ${isBuy ? 'bg-emerald-400' : 'bg-red-400'}`} />
   189→      <div>
   190→        <span className={`font-bold text-lg ${isBuy ? 'text-emerald-400' : 'text-red-400'}`}>{side}</span>
   191→        <span className="text-xs text-muted-foreground ml-2">{new Date(ts).toLocaleTimeString()}</span>
   192→      </div>
   193→      <Badge variant="outline" className="ml-auto text-xs">{(confidence * 100).toFixed(0)}%</Badge>
   194→    </div>
   195→  );
   196→}

   243→            </div>
   244→          </div>
   245→        </CardHeader>
