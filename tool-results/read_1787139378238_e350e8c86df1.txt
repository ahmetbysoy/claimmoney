    16→// --- Deterministic in-browser demo data ---
    17→function generateDemoCandles(n: number) {
    18→  const candles = [];
    19→  let price = 42000;
    20→  for (let i = 0; i < n; i++) {
    21→    const change = (Math.sin(i * 0.1) * 0.003) + (Math.cos(i * 0.05) * 0.002);
    22→    const open = price;
    23→    const close = price * (1 + change);
    24→    const high = Math.max(open, close) * (1 + Math.abs(Math.sin(i * 0.2)) * 0.002);
    25→    const low = Math.min(open, close) * (1 - Math.abs(Math.cos(i * 0.15)) * 0.002);
    26→    candles.push({ ts: 1000 + i * 60000, o: open, h: high, l: low, c: close, v: 100 + Math.sin(i) * 50 });
    27→    price = close;
    28→  }
    29→  return candles;
    30→}
    31→
    32→function computeStats(candles: { o: number; h: number; l: number; c: number; v: number }[]) {
    33→  const last = candles[candles.length - 1];
    34→  const prev = candles.length > 1 ? candles[candles.length - 2] : last;
    35→  const change = ((last.c - prev.c) / prev.c) * 100;
    36→  const high24 = Math.max(...candles.slice(-24).map(c => c.h));
    37→  const low24 = Math.min(...candles.slice(-24).map(c => c.l));
    38→  const totalVol = candles.slice(-24).reduce((s, c) => s + c.v, 0);
    39→  return { price: last.c, change, high24, low24, totalVol };
    40→}
