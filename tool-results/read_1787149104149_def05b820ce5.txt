   200→// SIGNAL DECAY BY TYPE (P1-16)
   201→// ═══════════════════════════════════════════════════════════════
   202→function decayRate(type: string): number {
   203→  switch (type) {
   204→    case 'WALL': case 'BID_WALL': case 'ASK_WALL': return 0.0002;
   205→    case 'SPOOFING': return 0.002;
   206→    case 'CASCADE_SQUEEZE': case 'CASCADE_CHAIN': case 'CASCADE_EXHAUSTED': return 0.003;
   207→    case 'CVD_BULL_DIV': case 'CVD_BEAR_DIV': return 0.003;
   208→    case 'VOID_ASK': case 'VOID_BID': return 0.0008;
   209→    case 'ICEBERG': case 'ICEBERG_FORMING': case 'ICEBERG_CONFIRMED': return 0.0003;
   210→    default: return 0.0005;
   211→  }
   212→}

   214→// ═══════════════════════════════════════════════════════════════
   215→// ADD SIGNAL (with confluence boost, VPIN penalty, cooldown)
   216→// ═══════════════════════════════════════════════════════════════
   217→function addSig(ty: string, bi: string, co: number, pr: number, ds: string, cf: number): void {
   218→  const t = now();
   219→  if (sLF[ty] && t - sLF[ty] < 8000) return;
   220→  sLF[ty] = t;
   221→  co = co < cMc ? 0 : co;
   222→  if (vVal > 0.7) co *= 0.7;
   223→  // Confluence boost: same-direction signals in 30s
   224→  let conf = 0;
   225→  for (let i = 0; i < Math.min(sC, 30); i++) {
   226→    const x = (sH - 1 - i + MS * 2) % MS;
   227→    if (t - sTm[x] > 30000) break;
   228→    if (sBi[x] === bi) conf++;
   229→  }
   230→  if (conf >= 2) co = cl(co * 1.3, 0, 98);
   231→
   232→  const x = sH % MS;
   233→  sTy[x] = ty; sBi[x] = bi; sDe[x] = Math.round(co);
   234→  sPr[x] = pr; sTm[x] = t; sDs[x] = ds; sCf[x] = cf || 0;
   235→  sVer[x] = 0; // pending verification
   236→  sH++; if (sC < MS) sC++;
   237→}

   240→// VERIFY SIGNALS (P2-6)
   241→// ═══════════════════════════════════════════════════════════════
   242→function verifySignals(): void {
   243→  const t = now();
   244→  for (let i = 0; i < Math.min(sC, 50); i++) {
   245→    const x = (sH - 1 - i + MS * 2) % MS;
   246→    if (sVer[x] !== 0) continue;
   247→    const age = t - sTm[x];
   248→    if (age < 5000) continue;
   249→    if (age > 30000) { sVer[x] = 2; continue; } // expired → miss
   250→    const dir = sBi[x];
   251→    const priceMove = mid > 0 ? (mid - sPr[x]) / sPr[x] * 100 : 0;
   252→    if ((dir === 'bullish' && priceMove > 0.03) || (dir === 'bearish' && priceMove < -0.03)) {
   253→      sVer[x] = 1; // verified hit
   254→    } else if ((dir === 'bullish' && priceMove < -0.03) || (dir === 'bearish' && priceMove > 0.03)) {
   255→      sVer[x] = 2; // verified miss
   256→    }
   257→  }
   258→}

   263→function calcConfluence(dir: string): number {

   269→    if (sBi[x] === dir) types.add(sTy[x]);
   270→  }
   271→  return types.size;
   272→}

   286→// P0: RISK GATE
   287→// ═══════════════════════════════════════════════════════════════
   288→function riskGate(): { pass: boolean; reason: string } {
   289→  const t = now();
   290→  if (t - lastBookT > 5000 && lastBookT > 0) return { pass: false, reason: 'Stale data' };
   291→  if (bN > 0 && aN > 0) {
   292→    const sp = (aP[0] - bP[0]) / mid * 1e4;
   293→    if (sp > 10) return { pass: false, reason: 'Spread > 10bps' };
   294→  }
   295→  if (vVal > 0.8) return { pass: false, reason: 'VPIN toxic' };
   296→  if (compAct) return { pass: false, reason: 'Compression active' };
   297→  if (cascadeChainLen >= 1 && t - cascadeChainT < 5000) return { pass: false, reason: 'Cascade active' };
   298→  return { pass: true, reason: '' };
   299→}

   302→// P0: PLAN STATE MACHINE
   303→// ═══════════════════════════════════════════════════════════════
   304→function updatePlanState(): void {
   305→  const gate = riskGate();
   306→  if (!gate.pass) {
   307→    if (planState !== 'NEUTRAL') { planState = 'NEUTRAL'; planId = ''; planDir = ''; planConf = 0; }
   308→    return;
   309→  }
   310→  const pl = genPlan();
   311→  if (!pl || pl.dir === 'NEUTRAL') {
   312→    if (planState !== 'NEUTRAL') { planState = 'NEUTRAL'; planId = ''; planDir = ''; planConf = 0; }
   313→    return;
   314→  }
   315→  const t = now();
   316→  if (pl.dir === lastPlanDir && t - lastPlanT < 30000) return;
   317→
   318→  const cf = calcConfluence(pl.dir);
   319→  if (cf < 3) {
   320→    if (planState !== 'NEUTRAL') { planState = 'NEUTRAL'; planId = ''; planDir = ''; planConf = 0; }
   321→    return;
   322→  }
   323→
   324→  const netRR = calcNetRR(pl);
   325→  if (netRR < 2.0) {
   326→    if (planState !== 'NEUTRAL') { planState = 'NEUTRAL'; planId = ''; planDir = ''; planConf = 0; }
   327→    return;
   328→  }
   329→
   330→  planState = 'CANDIDATE';
   331→  planDir = pl.dir; planConf = pl.conf; planEntry = pl.entry;
   332→  planSL = pl.sl; planTP = pl.tp; planRR = netRR;
   333→  planConfluence = cf; planTTL = cTTL * 1000;
   334→  planId = 'plan_' + t;
   335→  lastPlanDir = pl.dir; lastPlanT = t;
   336→}

   339→// P0: NET RR CALCULATOR
   340→// ═══════════════════════════════════════════════════════════════
   341→function calcNetRR(pl: { dir: string; entry: number; sl: number; tp: number; rr: number } | null): number {
   342→  if (!pl || pl.dir === 'NEUTRAL') return 0;
   343→  const grossRR = pl.rr;
   344→  const feeCost = (cFb * 2) / 10000;
   345→  const slipCost = 0.0005;
   346→  const totalCost = feeCost + slipCost;
   347→  const riskDist = Math.abs(pl.entry - pl.sl) / pl.entry;
   348→  const costRR = riskDist > 0 ? totalCost / riskDist : 0;
   349→  let net = grossRR - costRR;
   350→  if (fundingRate !== 0) {
   351→    const holdHrs = Math.min(cHold / 3600, 8);
   352→    net -= Math.abs(fundingRate) * holdHrs;
   353→  }
   354→  return net;
   355→}

   358→// PROC BOOK (P1-1: Velocity)
   359→// ═══════════════════════════════════════════════════════════════
   360→function procBook(bids: number[][], asks: number[][]): void {
   361→  const t = now();
   362→  // Save previous for velocity
   363→  pbN = bN; paN = aN;
   364→  for (let i = 0; i < bN; i++) { pbP[i] = bP[i]; pbQ[i] = bQ[i]; }
   365→  for (let i = 0; i < aN; i++) { paP[i] = aP[i]; paQ[i] = aQ[i]; }
   366→
   367→  bN = Math.min(bids.length, ML); aN = Math.min(asks.length, ML);
   368→  for (let i = 0; i < bN; i++) { bP[i] = bids[i][0]; bQ[i] = bids[i][1]; }
   369→  for (let i = 0; i < aN; i++) { aP[i] = asks[i][0]; aQ[i] = asks[i][1]; }
   370→  mid = bN > 0 && aN > 0 ? (bP[0] + aP[0]) * 0.5 : 0;
   371→
   372→  // Book velocity (P1-1)
   373→  if (lastBookT > 0) {
   374→    const dt = t - lastBookT;
   375→    if (dt > 0) {
   376→      let vel = 0;
   377→      for (let i = 0; i < Math.min(bN, pbN); i++) {
   378→        if (Math.abs(bP[i] - pbP[i]) < 0.001) vel += Math.abs(bQ[i] - pbQ[i]) * bP[i];
   379→      }
   380→      for (let i = 0; i < Math.min(aN, paN); i++) {
   381→        if (Math.abs(aP[i] - paP[i]) < 0.001) vel += Math.abs(aQ[i] - paQ[i]) * aP[i];
   382→      }
   383→      bookVelScore = vel / dt * 1000;
   384→    }
   385→  }
   386→  lastBookT = t;
   387→
   388→  calcMultiOBI();
   389→  updateLiqPools();
   390→  updateRegime();
   391→  runDet();
   392→  updFlow();
   393→  verifySignals();
   394→  updatePlanState();
   395→  postSt();
   396→}

   399→// MULTI-DEPTH OBI (P1-2)
   400→// ═══════════════════════════════════════════════════════════════
   401→function calcMultiOBI(): void {
   402→  const n5 = Math.min(5, bN, aN), n10 = Math.min(10, bN, aN), n20 = Math.min(20, bN, aN);
   403→  let bd5 = 0, ad5 = 0, bd10 = 0, ad10 = 0, bd20 = 0, ad20 = 0;
   404→  for (let i = 0; i < n5; i++) { bd5 += bP[i] * bQ[i]; ad5 += aP[i] * aQ[i]; }
   405→  for (let i = 0; i < n10; i++) { bd10 += bP[i] * bQ[i]; ad10 += aP[i] * aQ[i]; }
   406→  for (let i = 0; i < n20; i++) { bd20 += bP[i] * bQ[i]; ad20 += aP[i] * aQ[i]; }
   407→  const tot5 = bd5 + ad5, tot10 = bd10 + ad10, tot20 = bd20 + ad20;
   408→  obi5 = tot5 > 0 ? (bd5 - ad5) / tot5 : 0;
   409→  obi10 = tot10 > 0 ? (bd10 - ad10) / tot10 : 0;
   410→  obi20 = tot20 > 0 ? (bd20 - ad20) / tot20 : 0;
   411→  obiDiv = Math.abs(obi5 - obi20) > 0.3;
   412→}

   415→// LIQ POOL ESTIMATOR (P1-9)
   416→// ═══════════════════════════════════════════════════════════════
   417→function updateLiqPools(): void {
   418→  if (!mid) return;
   419→  const vwap = calcVWAP();
   420→  if (!vwap) return;
   421→  liqPoolLong = []; liqPoolShort = [];
   422→  const tiers = [10, 20, 50, 100];
   423→  for (const lev of tiers) {
   424→    const dist = vwap * (1 / lev);
   425→    liqPoolLong.push(vwap + dist);
   426→    liqPoolShort.push(vwap - dist);
   427→  }
   428→}

   430→function calcVWAP(): number {
   431→  if (tC < 10) return mid;
   432→  let sumPV = 0, sumV = 0;
   433→  const n = Math.min(tC, 100);
   434→  for (let i = 0; i < n; i++) {
   435→    const x = (tH - 1 - i + MT * 2) % MT;
   436→    sumPV += tP[x] * tQ[x]; sumV += tQ[x];
   437→  }
   438→  return sumV > 0 ? sumPV / sumV : mid;
   439→}

   442→// MARKET REGIME (P1-12)
   443→// ═══════════════════════════════════════════════════════════════
   444→function updateRegime(): void {
   445→  const sp = mid > 0 && bN > 0 && aN > 0 ? (aP[0] - bP[0]) / mid * 1e4 : 0;
   446→  const volProxy = sp * Math.max(tradeRate1s, 1);
   447→  volHist[volH % 60] = volProxy; volH++;
   448→  let dirProxy = 0;
   449→  if (cC >= 10) {
   450→    const cn = cV[(cH - 1 + MC) % MC], ct = cV[(cH - 10 + MC) % MC];
   451→    dirProxy = Math.abs(cn - ct);
   452→  }
   453→  dirHist[dirH % 60] = dirProxy; dirH++;
   454→
   455→  const avgVol = avg(volHist, Math.min(60, volH));
   456→  const avgDir = avg(dirHist, Math.min(60, dirH));
   457→
   458→  if (cascadeChainLen >= 2) { regime = 'CHAOS'; regimeScore = 90; }
   459→  else if (compAct && avgVol < 2) { regime = 'COILING'; regimeScore = 70; }
   460→  else if (avgVol > 5 && avgDir > 100000) { regime = 'TRENDING'; regimeScore = 80; }
   461→  else if (avgVol > 5 && avgDir < 50000) { regime = 'CHOPPY'; regimeScore = 60; }
   462→  else { regime = 'DEAD'; regimeScore = 30; }
   463→}

   466→// PROC TRADE (P1-3: Trade Rate, P1-4: Whale)
   467→// ═══════════════════════════════════════════════════════════════
   468→function procTrade(p: number, q: number, s: number, t: number): void {
   469→  const x = tH % MT;
   470→  tP[x] = p; tQ[x] = q; tS[x] = s; tT[x] = t; tN[x] = p * q;
   471→  // Tier classification (P1-4)
   472→  const not = p * q;
   473→  let tier = 0;
   474→  if (not > 1e6) { tier = 3; whaleT3++; }
   475→  else if (not > 2e5) { tier = 2; whaleT2++; }
   476→  else if (not > 5e4) { tier = 1; whaleT1++; }
   477→  tTier[x] = tier;
   478→  tH++; if (tC < MT) tC++;
   479→
   480→  updateTradeRates(t, not);
   481→
   482→  // CVD
   483→  const d = s === 0 ? not : -not; cvd += d;
   484→  const ci = cH % MC; cT[ci] = t; cV[ci] = cvd; cH++; if (cC < MC) cC++;
   485→
   486→  // VPIN
   487→  if (s === 0) vBB += not; else vSB += not; vBN += not;
   488→  if (vBN >= vBucketSz) {
   489→    const tot = vBB + vSB;
   490→    if (tot > 0) { vSum += Math.abs(vBB - vSB) / tot; vCnt++; if (vCnt > 50) vCnt = 50; vVal = vSum / vCnt; }
   491→    vBB = 0; vSB = 0; vBN = 0;
   492→  }
   493→
   494→  // Flow bucket
   495→  if (s === 0) fbB += not; else fbSe += not; fbA += not;
   496→  if (fbPH < p) fbPH = p; if (fbPL > p || fbPL === 0) fbPL = p;
   497→
   498→  chkIceberg(p, not, s);
   499→}

   504→function updateTradeRates(t: number, not: number): void {
   505→  const nowT = now();

   512→    if (age < 5000) { c5++; n5 += tN[x]; }
   513→    if (age < 30000) { c30++; }
   514→  }
   515→  tradeRate1s = c1; tradeRate5s = c5; tradeRate30s = c30;
   516→  const avg5s = c5 > 0 ? n5 / 5 : 0;
   517→  tapeSpike = c1 > avg5s * 3 && c1 >= 5;
   518→  if (tapeSpike) {
   519→    addSig('TAPE_SPIKE', c1 > n1 ? 'bullish' : 'bearish', 65, mid, 'Tape speed spike: ' + c1 + ' trades/s', 0);
   520→  }
   521→}

   526→function procLiq(s: number, p: number, q: number, t: number): void {

   552→// CASCADE EXHAUSTION (P1-8)
   553→// ═══════════════════════════════════════════════════════════════
   554→function chkCascadeExhaustion(): void {
   555→  if (cascadeChainLen < 3) return;
   556→  const dir = cascadeChainDir;
   557→  const arr = dir === 1 ? bP : aP;
   558→  const qty = dir === 1 ? bQ : aQ;

   562→  let near = 0;
   563→  for (let i = 0; i < cnt; i++) {
   564→    if ((dir === 1 && arr[i] >= thr) || (dir === 0 && arr[i] <= thr)) near += arr[i] * qty[i];
   565→  }
   566→  const lastLiq = lN[(lH - 1 + MQ) % MQ];
   567→  if (near > lastLiq * 1.2) {
   568→    addSig('CASCADE_EXHAUSTED', dir === 1 ? 'bullish' : 'bearish', 85, mid, 'Cascade exhausted → reversal', 0);
   569→    cascadeChainLen = 0;
   570→  }

   574→// ICEBERG (P2-2: Lifecycle)
   575→// ═══════════════════════════════════════════════════════════════
   576→function chkIceberg(price: number, notional: number, side: number): void {
   577→  if (notional < cIb) return;
   578→  const arr = side === 0 ? aP : bP;
   579→  const qty = side === 0 ? aQ : bQ;
   580→  const cnt = side === 0 ? aN : bN;
   581→  for (let i = 0; i < Math.min(cnt, 20); i++) {
   582→    if (Math.abs(arr[i] - price) / price < 0.0002) {
   583→      iTr[i] += notional; iDp[i] = qty[i]; iCt[i]++;
   584→      if (iState[i] === 0) { iState[i] = 1; iT[i] = now(); } // FORMING
   585→      else if (iState[i] === 1 && iCt[i] >= 3) { iState[i] = 2; } // CONFIRMED
   586→
   587→      if (iCt[i] >= 3) {
   588→        const avgT = iTr[i] / iCt[i];
   589→        const avgP = mid || 1;
   590→        const tQty = avgT / avgP;
   591→        const sc = tQty / Math.max(iDp[i], 0.001);
   592→        if (sc > 2.5) {
   593→          const st = iState[i] === 2 ? 'CONFIRMED' : 'FORMING';
   594→          addSig('ICEBERG_' + st, side === 0 ? 'bullish' : 'bearish',
   595→            cl(60 + sc * 5, 60, 88), price,
   596→            'Iceberg ' + st + ' @ ' + price.toFixed(2) + ' ratio:' + sc.toFixed(1) + 'x', 0);
   597→          if (iState[i] === 2) { iState[i] = 4; iCt[i] = 0; } // CONSUMED
   598→        }
   599→      }
   600→      break;
   601→    }
   602→  }
   603→}

   606→// CASCADE (P1-7)
   607→// ═══════════════════════════════════════════════════════════════
   608→function chkCascade(side: number, price: number, notional: number): void {
   609→  if (notional < 100000) return;
   610→  const arr = side === 1 ? bP : aP;
   611→  const qty = side === 1 ? bQ : aQ;
   612→  const cnt = side === 1 ? bN : aN;
   613→  const thr = side === 1 ? price * 0.999 : price * 1.001;
   614→  let near = 0;
   615→  for (let i = 0; i < cnt; i++) {
   616→    if ((side === 1 && arr[i] >= thr) || (side === 0 && arr[i] <= thr)) near += arr[i] * qty[i];
   617→  }
   618→  if (notional > near * 1.5) {
   619→    addSig('CASCADE_SQUEEZE', side === 1 ? 'bearish' : 'bullish', 88, price,
   620→      'Cascade: $' + (notional / 1e6).toFixed(2) + 'M vs depth $' + (near / 1e6).toFixed(2) + 'M', 0);
   621→  }
   622→}

   644→// FLOW (P1-M: Trend, P1-N: Exhaustion)

   646→function updFlow(): void {
   647→  const t = now();
   648→  if (fbS === 0) { fbS = t; fbPO = mid; fbPH = mid; fbPL = mid; return; }
   649→  const el = t - fbS;
   650→  const close = cFm === 'time' ? el >= cFt : fbA >= cFv;
   651→  if (close && fbA > 0) {
   652→    const d = fbB - fbSe;
   653→    const pr = cl(d / fbA * 100, -100, 100);
   654→    const st = cl(Math.abs(d) / fbA * 100, 0, 100);
   655→    const fi = fH % MF;
   656→    fPr[fi] = pr; fSt[fi] = st; fDe[fi] = d;
   657→    fBu[fi] = fbB; fSe[fi] = fbSe; fTm[fi] = fbS; fLq[fi] = fbL;
   658→    fH++; if (fC < MF) fC++;
   659→
   660→    // Absorption
   661→    if (Math.abs(d) > 100000 && fbPH > 0 && fbPL > 0 && Math.abs(fbPH - fbPL) / mid < 0.0005) {
   662→      absInfo = { ts: t, side: d > 0 ? 'bid' : 'ask', delta: d, price: mid };
   663→      addSig('ABSORPTION', d > 0 ? 'bullish' : 'bearish', 75, mid,
   664→        'Absorption: delta $' + (d / 1e3).toFixed(0) + 'K zero move', 0);
   665→    }
   666→
   667→    detFlowTrend();
   668→
   669→    fbS = t; fbB = 0; fbSe = 0; fbA = 0; fbL = 0;
   670→    fbPO = mid; fbPH = mid; fbPL = mid;
   671→    detFlow();
   672→  }
   673→}

   675→function detFlowTrend(): void {
   676→  if (fC < 5) return;
   677→  let pos = 0, neg = 0;
   678→  for (let i = 0; i < Math.min(fC, 5); i++) {
   679→    const x = (fH - 1 - i + MF * 2) % MF;
   680→    if (fDe[x] > 0) pos++; else if (fDe[x] < 0) neg++;
   681→  }
   682→  if (pos === 5) addSig('FLOW_SUSTAINED', 'bullish', 60, mid, 'Sustained buying pressure', 0);
   683→  else if (neg === 5) addSig('FLOW_SUSTAINED', 'bearish', 60, mid, 'Sustained selling pressure', 0);
   684→
   685→  // Exhaustion (P1-N)
   686→  const la = (fH - 1 + MF) % MF;
   687→  if (Math.abs(fPr[la]) > 80 && fC >= 3) {
   688→    const prev = fPr[(fH - 2 + MF) % MF];
   689→    if (Math.abs(fPr[la]) > Math.abs(prev) * 0.8) {
   690→      addSig('FLOW_EXHAUSTION', fDe[la] > 0 ? 'bearish' : 'bullish', 70, mid, 'Flow exhaustion detected', 0);
   691→    }
   692→  }

   720→// WALLS (P1-5: Dynamic Threshold, P1-6: Execution Ratio)
   721→// ═══════════════════════════════════════════════════════════════
   722→function detWalls(): void {
   723→  const t = now();
   724→  const n10 = Math.min(10, bN, aN);
   725→  const bq: number[] = [], aq: number[] = [];
   726→  for (let i = 0; i < n10; i++) { bq[i] = bQ[i]; aq[i] = aQ[i]; }
   727→  const mB = med(new Float64Array(bq), n10);
   728→  const mA = med(new Float64Array(aq), n10);
   729→
   730→  // Dynamic threshold (P1-5)
   731→  const sp = mid > 0 ? (aP[0] - bP[0]) / mid * 1e4 : 0;
   732→  const volFactor = cl(sp / 5, 0.5, 3.0);
   733→  const thrB = mB * cWm * volFactor, thrA = mA * cWm * volFactor;
   734→
   735→  // Bid walls
   736→  const nBP: number[] = [], nBQ: number[] = [], nBI: number[] = [];
   737→  const nBL: number[] = [], nBX: number[] = [], nBE: number[] = [];
   738→  const lim = Math.min(bN, 15);
   739→  for (let i = 0; i < lim; i++) {
   740→    if (bQ[i] > thrB) {
   741→      let ex = -1;
   742→      for (let j = 0; j < wBC; j++) {
   743→        if (Math.abs(wBP[j] - bP[i]) / bP[i] < 0.0001) { ex = j; break; }
   744→      }
   745→      if (ex >= 0) {
   746→        const qd = wBQ[ex] - bQ[i];
   747→        nBP.push(bP[i]); nBQ.push(bQ[i]); nBI.push(wBI[ex]);
   748→        nBL.push(t); nBX.push(qd > wBI[ex] * 0.5 ? 1 : 0); nBE.push(wBE[ex]);
   749→      } else {
   750→        nBP.push(bP[i]); nBQ.push(bQ[i]); nBI.push(bQ[i]);
   751→        nBL.push(t); nBX.push(0); nBE.push(0);
   752→        if (bQ[i] * bP[i] > 100000) {
   753→          addSig('BID_WALL', 'bullish', cl(55 + (bQ[i] / mB) * 4, 55, 92), bP[i],
   754→            'Bid wall ' + bP[i].toFixed(2) + ' ' + bQ[i].toFixed(3), 0);
   755→        }
   756→      }
   757→    }
   758→  }
   759→  wBC = nBP.length;
   760→  for (let i = 0; i < wBC; i++) {
   761→    wBP[i] = nBP[i]; wBQ[i] = nBQ[i]; wBI[i] = nBI[i];
   762→    wBL[i] = nBL[i]; wBX[i] = nBX[i]; wBE[i] = nBE[i];
   763→  }
   764→
   765→  // Ask walls
   766→  const nAP: number[] = [], nAQ: number[] = [], nAI: number[] = [];
   767→  const nAL: number[] = [], nAX: number[] = [], nAE: number[] = [];
   768→  const limA = Math.min(aN, 15);
   769→  for (let i = 0; i < limA; i++) {
   770→    if (aQ[i] > thrA) {
   771→      let ex = -1;
   772→      for (let j = 0; j < wAC; j++) {
   773→        if (Math.abs(wAP[j] - aP[i]) / aP[i] < 0.0001) { ex = j; break; }
   774→      }
   775→      if (ex >= 0) {
   776→        const qd = wAQ[ex] - aQ[i];
   777→        nAP.push(aP[i]); nAQ.push(aQ[i]); nAI.push(wAI[ex]);
   778→        nAL.push(t); nAX.push(qd > wAI[ex] * 0.5 ? 1 : 0); nAE.push(wAE[ex]);
   779→      } else {
   780→        nAP.push(aP[i]); nAQ.push(aQ[i]); nAI.push(aQ[i]);
   781→        nAL.push(t); nAX.push(0); nAE.push(0);
   782→        if (aQ[i] * aP[i] > 100000) {
   783→          addSig('ASK_WALL', 'bearish', cl(55 + (aQ[i] / mA) * 4, 55, 92), aP[i],
   784→            'Ask wall ' + aP[i].toFixed(2) + ' ' + aQ[i].toFixed(3), 0);
   785→        }
   786→      }
   787→    }
   788→  }
   789→  wAC = nAP.length;
   790→  for (let i = 0; i < wAC; i++) {
   791→    wAP[i] = nAP[i]; wAQ[i] = nAQ[i]; wAI[i] = nAI[i];
   792→    wAL[i] = nAL[i]; wAX[i] = nAX[i]; wAE[i] = nAE[i];
   793→  }
   794→
   795→  // Wall execution ratio (P1-6)
   796→  for (let i = 0; i < wBC; i++) {
   797→    if (wBQ[i] > 0 && wBE[i] / wBQ[i] > 0.7) {
   798→      addSig('WALL_CONSUMED', 'bullish', 75, wBP[i], 'Bid wall consumed @ ' + wBP[i].toFixed(2), 0);

   800→  }
   801→  for (let i = 0; i < wAC; i++) {
   802→    if (wAQ[i] > 0 && wAE[i] / wAQ[i] > 0.7) {
   803→      addSig('WALL_CONSUMED', 'bearish', 75, wAP[i], 'Ask wall consumed @ ' + wAP[i].toFixed(2), 0);
   804→    }
   805→  }

   937→function genPlan(): { dir: string; conf: number; entry: number; sl: number; tp: number; rr: number; confluence: number } | null {
   938→  const t = now();
   939→  let bs = 0, be = 0;
   940→  for (let i = 0; i < Math.min(sC, 30); i++) {
   941→    const x = (sH - 1 - i + MS * 2) % MS;
   942→    if (t - sTm[x] > 30000) break;
   943→    if (sBi[x] === 'bullish') bs += sDe[x];
   944→    else if (sBi[x] === 'bearish') be += sDe[x];
   945→  }
   946→  if (!mid) return { dir: 'NEUTRAL', conf: 0, entry: 0, sl: 0, tp: 0, rr: 0, confluence: 0 };
   947→
   948→  let dir: string = 'NEUTRAL', conf = 0;
   949→  if (bs > be + 50) { dir = 'LONG'; conf = cl(50 + (bs - be) / 5, 50, 95); }
   950→  else if (be > bs + 50) { dir = 'SHORT'; conf = cl(50 + (be - bs) / 5, 50, 95); }
   951→  if (dir === 'NEUTRAL') return null;
   952→
   953→  const cf = calcConfluence(dir);
   954→  if (cf >= 4) conf = cl(conf + 15, 0, 98);
   955→  if (cf >= 5) conf = cl(conf + 25, 0, 98);
   956→
   957→  let sl: number, tp: number;
   958→  if (dir === 'LONG') {
   959→    let wallSl: number | null = null;
   960→    for (let i = 0; i < wAC; i++) { wallSl = wAP[i]; break; }
   961→    sl = wallSl ? wallSl * 0.999 : mid - (aP[0] - bP[0]) * 7.5;
   962→    let voidTp: number | null = null;
   963→    for (let i = 0; i < Math.min(sC, 15); i++) {
   964→      const x = (sH - 1 - i + MS * 2) % MS;
   965→      if (t - sTm[x] > 30000) break;
   966→      if (sTy[x] === 'VOID_ASK') { voidTp = sPr[x]; break; }
   967→    }
   968→    const risk = Math.abs(mid - sl);
   969→    tp = voidTp && Math.abs(voidTp - mid) / risk >= 2.5 ? voidTp : mid + risk * 2.5;
   970→  } else {
   971→    let wallSl: number | null = null;
   972→    for (let i = 0; i < wBC; i++) { wallSl = wBP[i]; break; }
   973→    sl = wallSl ? wallSl * 1.001 : mid + (aP[0] - bP[0]) * 7.5;
   974→    let voidTp: number | null = null;
   975→    for (let i = 0; i < Math.min(sC, 15); i++) {
   976→      const x = (sH - 1 - i + MS * 2) % MS;
   977→      if (t - sTm[x] > 30000) break;
   978→      if (sTy[x] === 'VOID_BID') { voidTp = sPr[x]; break; }
   979→    }
   980→    const risk = Math.abs(sl - mid);
   981→    tp = voidTp && Math.abs(voidTp - mid) / risk >= 2.5 ? voidTp : mid - risk * 2.5;
   982→  }
   983→  const risk = Math.abs(mid - sl);
   984→  const rr = risk > 0 ? Math.abs(tp - mid) / risk : 0;
   985→  if (rr < 2.5) return null;
   986→
   987→  sl = normPrice(sl); tp = normPrice(tp);
   988→  return { dir, conf: Math.round(conf), entry: mid, sl, tp, rr, confluence: cf };
   989→}

   991→// ═══════════════════════════════════════════════════════════════
   992→// CALC MICRO (P1-U: True Kelly, P1-V: Confidence-adjusted)
   993→// ═══════════════════════════════════════════════════════════════
   994→function calcMicro(pl: { dir: string; entry: number; sl: number; tp: number; rr: number; conf: number } | null): Record<string, number | string> | null {
   995→  if (!pl || pl.dir === 'NEUTRAL') return null;
   996→  const maxR = cBal * (cRp / 100);
   997→  const rpu = Math.abs(pl.entry - pl.sl);
   998→  if (!rpu) return null;
   999→
  1000→  // True Kelly (P1-U)
  1001→  let W = 0.55; // default win rate
  1002→  const R = pl.rr;
  1003→  let kelly = W - (1 - W) / R;
  1004→  if (kelly < 0) return null;
  1005→
  1006→  // Confidence-adjusted (P1-V)
  1007→  kelly = kelly * (pl.conf / 100);
  1008→
  1009→  // Hard risk cap (P1-W)
  1010→  const riskPct = Math.min(kelly * 100 / 2, cRp);
  1011→  const actualRisk = cBal * (riskPct / 100);
  1012→  const actualQty = actualRisk / rpu;
  1013→  const actualNot = actualQty * pl.entry;
  1014→
  1015→  let lev = actualNot / cBal;
  1016→  lev = Math.min(lev, cMl);
  1017→  lev = Math.max(lev, 1);
  1018→  const mar = actualNot / lev;
  1019→  const fee = actualNot * (cFb / 10000) * 2;
  1020→  const be = actualQty > 0 ? fee / actualQty : 0;
  1021→
  1022→  // Tier-based maintenance margin (P1-X)
  1023→  let mmr = 0.004;
  1024→  if (actualNot > 50000) mmr = 0.005;
  1025→  if (actualNot > 250000) mmr = 0.01;
  1026→  if (actualNot > 1000000) mmr = 0.025;
  1027→
  1028→  const liq = pl.dir === 'LONG'
  1029→    ? pl.entry * (1 - 1 / lev + mmr)
  1030→    : pl.entry * (1 + 1 / lev - mmr);
  1031→
  1032→  const fundCost = fundingRate !== 0 ? Math.abs(fundingRate) * actualNot : 0;
  1033→  const ordT = vVal > 0.7 ? 'LIMIT ONLY' : 'MARKET';
  1034→
  1035→  return {
  1036→    riskPct: Number(riskPct.toFixed(2)),
  1037→    notional: actualNot,
  1038→    margin: mar,
  1039→    lev: Number(lev.toFixed(1)),
  1040→    be,
  1041→    liq,
  1042→    kelly: Number((kelly * 100).toFixed(1)),
  1043→    ordT,
  1044→    fundCost,
  1045→  };
  1046→}

  1049→// NARRATIVE (P2-W, P2-X, P2-Y)
  1050→// ═══════════════════════════════════════════════════════════════
  1051→function genNarrative(pl: { dir: string; conf: number; rr: number; confluence: number } | null): string {
  1052→  if (!pl || pl.dir === 'NEUTRAL') {
  1053→    return '🟡 ' + regime + ' | Piyasa net yonlu sinyal uretmiyor. Bekleme modunda.';
  1054→  }
  1055→  const t = now();
  1056→  let parts: string[] = [], evidence: string[] = [];
  1057→  for (let i = 0; i < Math.min(sC, 10); i++) {
  1058→    const x = (sH - 1 - i + MS * 2) % MS;
  1059→    if (t - sTm[x] > 30000) break;
  1060→    if (sTy[x].includes('WALL')) { parts.push((sTy[x].includes('BID') ? 'Bid' : 'Ask') + ' duvari @ ' + sPr[x].toFixed(2)); evidence.push('WALL'); }
  1061→    if (sTy[x].includes('CVD')) { parts.push('CVD ' + (sBi[x] === 'bullish' ? 'pozitif' : 'negatif') + ' uyumsuzluk'); evidence.push('CVD'); }
  1062→    if (sTy[x].includes('SPOOF')) { parts.push('Spoofing tespit edildi'); evidence.push('SPOOF'); }
  1063→    if (sTy[x].includes('ABSORPTION') || sTy[x].includes('COMPRESSION')) { parts.push('Emilim/sikisma bolgesi'); evidence.push('ABSORB'); }
  1064→    if (sTy[x].includes('CASCADE')) { parts.push('Cascade aktivitesi'); evidence.push('CASCADE'); }
  1065→    if (sTy[x].includes('ICEBERG')) { parts.push('Iceberg tespiti'); evidence.push('ICEBERG'); }
  1066→  }
  1067→  if (!parts.length) parts.push(Math.min(sC, 30) + ' sinyal konverjans');
  1068→
  1069→  const regimeEmoji = regime === 'TRENDING' ? '🟢' : regime === 'CHOPPY' ? '🟠' : regime === 'COILING' ? '🔵' : regime === 'CHAOS' ? '🔴' : '⚪';
  1070→  return regimeEmoji + ' ' + regime + ' | Fiyat ' + (pl.dir === 'LONG' ? 'yukari' : 'asagi') + ' yonlu baski: ' + parts.join('; ') + '. RR: ' + pl.rr.toFixed(1) + '. Confluence: ' + pl.confluence + '/6';
  1071→}

  1074→// POST STATE (P1-Z: Throttled, P2-AA: Diff-based)
  1075→// ═══════════════════════════════════════════════════════════════
  1076→function postSt(): void {
  1077→  const t = now();
  1078→  if (t - lastPostT < 200) return; // Throttle 200ms
  1079→  lastPostT = t;
  1080→
  1081→  const pl = genPlan();
  1082→  const mi = calcMicro(pl);
  1083→  const nar = genNarrative(pl);
  1084→
  1085→  // Signal decay by type (P1-16)
  1086→  const sigs: Record<string, unknown>[] = [];
  1087→  for (let i = 0; i < Math.min(sC, 30); i++) {
  1088→    const x = (sH - 1 - i + MS * 2) % MS;
  1089→    if (t - sTm[x] > 120000) break;
  1090→    const age = t - sTm[x];
  1091→    const dr = decayRate(sTy[x]);
  1092→    const decay = Math.exp(-dr * age);
  1093→    sigs.push({
  1094→      t: sTy[x], b: sBi[x], c: Math.round(sDe[x] * decay),
  1095→      p: sPr[x], ts: sTm[x], d: sDs[x], cf: sCf[x], ver: sVer[x],
  1096→    });
  1097→  }
  1098→
  1099→  const flows: Record<string, number>[] = [];
  1100→  for (let i = 0; i < Math.min(fC, 30); i++) {
  1101→    const x = (fH - 1 - i + MF * 2) % MF;
  1102→    flows.unshift({ p: fPr[x], s: fSt[x], d: fDe[x], b: fBu[x], sl: fSe[x] });
  1103→  }
  1104→
  1105→  const cvds: number[] = [];
  1106→  for (let i = 0; i < Math.min(cC, 60); i++) {
  1107→    const x = (cH - 1 - i + MC * 2) % MC;
  1108→    cvds.unshift(cV[x]);
  1109→  }
  1110→
  1111→  let ll = 0, ls = 0;
  1112→  for (let i = 0; i < lC; i++) {
  1113→    const x = (lH - 1 - i + MQ * 2) % MQ;
  1114→    if (t - lT[x] > 60000) break;
  1115→    if (lS[x] === 1) ll++; else ls++;
  1116→  }
  1117→
  1118→  let bd = 0, ad = 0;
  1119→  const n10 = Math.min(10, bN, aN);
  1120→  for (let i = 0; i < n10; i++) { bd += bP[i] * bQ[i]; ad += aP[i] * aQ[i]; }
  1121→  const obi = n10 > 0 ? (bd - ad) / (bd + ad) : 0;
  1122→
  1123→  let mp = 0, wSum = 0, qSum = 0;
  1124→  const n5 = Math.min(5, bN, aN);
  1125→  for (let i = 0; i < n5; i++) { wSum += bP[i] * bQ[i]; qSum += bQ[i]; wSum += aP[i] * aQ[i]; qSum += aQ[i]; }
  1126→  mp = qSum > 0 ? wSum / qSum : mid;
  1127→  const spBps = mid > 0 && bN > 0 && aN > 0 ? (aP[0] - bP[0]) / mid * 1e4 : 0;
  1128→
  1129→  postMessage({
  1130→    type: 'st', mid, vpin: vVal, cvd, cvds, flows, sigs, plan: pl, micro: mi, narrative: nar,
  1131→    planState, planId, planDir, planConf, planEntry, planSL, planTP, planRR,
  1132→    planConfluence, planTTL: planTTL > 0 ? Math.round(planTTL / 1000) : 0,
  1133→    regime, regimeScore, obi, obi5, obi20, obiDiv, bookVelScore,
  1134→    liqLong: ll, liqShort: ls, mp, spBps, absorption: absInfo ? { ...absInfo } : null,
  1135→    tapeSpike, tradeRate1s, whaleT1, whaleT3,
  1136→    bN, aN,
  1137→    bidP: Array.from(bP.subarray(0, bN)), bidQ: Array.from(bQ.subarray(0, bN)),
  1138→    askP: Array.from(aP.subarray(0, aN)), askQ: Array.from(aQ.subarray(0, aN)),
  1139→    wBC, wBP: Array.from(wBP.subarray(0, wBC)), wBX: Array.from(wBX.subarray(0, wBC)),
  1140→    wBE: Array.from(wBE.subarray(0, wBC)),
  1141→    wAC, wAP: Array.from(wAP.subarray(0, wAC)), wAX: Array.from(wAX.subarray(0, wAC)),
  1142→    wAE: Array.from(wAE.subarray(0, wAC)),
  1143→    liqPoolLong, liqPoolShort,
  1144→    cascadeChainLen, cascadeChainDir,
  1145→  });
  1146→  absInfo = null;
  1147→}
