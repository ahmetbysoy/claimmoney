   646→function riskGate(){

   667→function updatePlanState(){

   688→    return;
   689→  }
   690→  
   691→  // net_RR check
   692→  const netRR=calcNetRR(pl);

   705→function calcNetRR(pl){
   706→  if(!pl||pl.dir==='NEUTRAL')return 0;
   707→  const grossRR=pl.rr;
   708→  const feeCost=(cFb*2)/10000;
   709→  const slipCost=0.0005; // avg 5bps slippage
   710→  const totalCost=feeCost+slipCost;
   711→  const riskDist=Math.abs(pl.entry-pl.sl)/pl.entry;
   712→  const costRR=totalCost/riskDist;
   713→  let net=grossRR-costRR;
   714→  // Funding adjustment
   715→  if(fundingRate!==0){
   716→    const holdHrs=Math.min(cHold/3600,8);
   717→    net-=Math.abs(fundingRate)*holdHrs;
   718→  }
   719→  return net;
   720→}

  1214→/* ── CALC MICRO (P1-U: True Kelly, P1-V: Confidence-adjusted) ── */
  1215→function calcMicro(pl){
  1216→  if(!pl||pl.dir==='NEUTRAL')return null;
  1217→  const maxR=cBal*(cRp/100);const rpu=Math.abs(pl.entry-pl.sl);if(!rpu)return null;
  1218→  const qty=maxR/rpu;const not=qty*pl.entry;
  1219→  
  1220→  // True Kelly (P1-U)
  1221→  let W=0.55; // default, will be overridden by paper stats
  1222→  const R=pl.rr;
  1223→  let kelly=W-(1-W)/R;
  1224→  if(kelly<0)return null; // Kelly says don't trade
  1225→  
  1226→  // Confidence-adjusted (P1-V)
  1227→  kelly=kelly*(pl.conf/100);
  1228→  
  1229→  // Hard risk cap (P1-W)
  1230→  const riskPct=Math.min(kelly*100/2,cRp); // Half-Kelly, capped at base risk
  1231→  const actualRisk=cBal*(riskPct/100);
  1232→  const actualQty=actualRisk/rpu;
  1233→  const actualNot=actualQty*pl.entry;
  1234→  
  1235→  let lev=actualNot/cBal;lev=Math.min(lev,cMl);lev=Math.max(lev,1);
  1236→  const mar=actualNot/lev;
  1237→  const fee=actualNot*(cFb/10000)*2;
  1238→  const be=fee/actualQty;
  1239→  
  1240→  // Tier-based maintenance margin (P1-X)
  1241→  let mmr=0.004;
  1242→  if(actualNot>50000)mmr=0.005;
  1243→  if(actualNot>250000)mmr=0.01;
  1244→  if(actualNot>1000000)mmr=0.025;
  1245→  
  1246→  const liq=pl.dir==='LONG'?pl.entry*(1-1/lev+mmr):pl.entry*(1+1/lev-mmr);
  1247→  
  1248→  // Funding (P1-Y)
  1249→  const fundCost=fundingRate!==0?Math.abs(fundingRate)*actualNot:0;
  1250→  
  1251→  const ordT=vVal>.7?'LIMIT ONLY':'MARKET';
  1252→  return{riskPct:riskPct.toFixed(2),notional:actualNot,margin:mar,lev:lev.toFixed(1),be,liq,kelly:(kelly*100).toFixed(1),ordT,fundCost};
  1253→}
